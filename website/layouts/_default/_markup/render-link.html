{{- $u := urls.Parse .Destination -}} {{- $href := $u.String -}} {{- if not $u.IsAbs -}} {{- /* Try to get the page based on the path */ -}} {{- $path := strings.TrimSuffix ".md" $u.Path -}} {{- if strings.HasSuffix $path "_index" -}} {{- $path = strings.TrimSuffix "_index" $path -}} {{- if eq $path "" -}} {{- $path = "." -}} {{- else -}} {{- $path = strings.TrimSuffix "/" $path -}} {{- end -}} {{- end -}} {{- with .Page.GetPage $path -}} {{- $href = .RelPermalink -}} {{- if $u.Fragment -}} {{- $href = printf "%s#%s" $href $u.Fragment -}} {{- end -}} {{- else -}} {{- /* If GetPage fails, do a simple text replacement as fallback */ -}} {{- if strings.HasSuffix $u.Path ".md" -}} {{- $href = replace $href ".md" "/" -}} {{- end -}} {{- end -}} {{- end -}} {{ .Text | safeHTML }}
